בדיקת הלוג (ב-Supabase ← SQL Editor)

-- כל השורות של הריצות הסינתטיות האחרונות
select sid, kind, turn, input_mode, word_count, has_hebrew, q_type, idle_level, created_at
from public.story_logs where sid like 'qa-%' order by created_at;

-- שורות turn שחסר בהן שדה (צריך להחזיר 0)
select count(*) from public.story_logs
where kind = 'turn' and (word_count is null or has_hebrew is null or q_type is null or input_mode is null);

-- אין עמודת טקסט בטבלה (צריך להחזיר 0 שורות)
select column_name from information_schema.columns
where table_name = 'story_logs' and column_name in ('child_text','juno_text','message','text');

-- לכל ריצה יש start ו-end (צריך להחזיר 12 שורות: 6 דמויות × 2 מודלים, עם 1 ו-1)
select sid, count(*) filter (where kind='start') as starts, count(*) filter (where kind='end') as ends
from public.story_logs where sid like 'qa-%' group by sid;

-- ניקוי הריצות הסינתטיות
delete from public.story_logs where sid like 'qa-%';
